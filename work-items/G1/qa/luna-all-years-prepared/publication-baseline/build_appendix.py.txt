#!/usr/bin/env python3
"""Generate the English G1 appendix and native tables from frozen Stata aggregates.

This publication builder never fits an estimator or reads a firm-level dataset.
Every empirical table cell and numerical prose token records its exact operands.
"""
from __future__ import annotations
import argparse,csv,hashlib,json,math,re
from pathlib import Path
import numpy as np
import pandas as pd

P=Path(__file__).resolve().parent
ROOT=P.parents[2]
DEFAULT=P.parent/'release'/'outputs'
METHOD={'op':'OP','lp':'LP','acf':'ACF','wrdg':'Wooldridge'}
SOURCE="G0.5 v3 modified manufacturing records with fictional company links; authors' calculations."


def sha(p): return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def clean(v):
    if isinstance(v,(np.integer,np.floating)): v=v.item()
    return None if isinstance(v,float) and not math.isfinite(v) else v

def year_text(values):
    """Compact calendar ranges while preserving gaps in the supplied years."""
    years=sorted({int(v) for v in values if pd.notna(v) and int(v)>0})
    groups=[]
    for year in years:
        if groups and year==groups[-1][-1]+1: groups[-1].append(year)
        else: groups.append([year])
    return ', '.join(str(g[0]) if len(g)==1 else f'{g[0]}--{g[-1]}' for g in groups)

def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--results',type=Path,default=DEFAULT)
    parser.add_argument('--deep-results',type=Path,default=P.parent/'deep-release'/'outputs')
    a=parser.parse_args()
    src=a.results.resolve()
    if (src/'outputs').is_dir(): src=src/'outputs'
    specpath=P.parent/'nominal-specification.json'
    spec=json.loads(specpath.read_text())
    traces={'table_cells':[],'numerical_prose':[]}
    inputs={}; raw={}; data={}
    for name in ['nominal_sample_support','nominal_coefficients','nominal_bootstrap_summary','nominal_distribution_quantiles','nominal_pairwise_comparisons']:
        path=src/(name+'.csv')
        data[name]=pd.read_csv(path,dtype={'sector':str})
        with path.open(newline='') as f: raw[name]=list(csv.DictReader(f))
        inputs[name]={'path':str(path),'sha256':sha(path),'rows':len(data[name])}
    deep_src=a.deep_results.resolve()
    if (deep_src/'outputs').is_dir(): deep_src=deep_src/'outputs'
    for path in sorted(deep_src.glob('deep_*.csv')):
        name=path.stem
        data[name]=pd.read_csv(path,dtype={'sector':str})
        with path.open(newline='') as stream: raw[name]=list(csv.DictReader(stream))
        inputs[name]={'path':str(path.resolve()),'sha256':sha(path),'rows':len(data[name])}
    assert len([k for k in data if k.startswith('deep_')])==12
    f=data['nominal_coefficients']; b=data['nominal_bootstrap_summary']; d=data['nominal_distribution_quantiles']; pairs=data['nominal_pairwise_comparisons']; supp=data['nominal_sample_support']
    primary=f[(f.variant=='primary')&(f.status=='fit_success')&(f.sample_id=='own')]
    sectors=sorted(primary.sector.unique())
    assert not primary.empty, 'No admitted own-sample estimates to publish'
    methods=[m for m in METHOD if m in set(primary.method)]
    common_primary=f[(f.variant=='primary')&(f.status=='fit_success')&(f.sample_id=='common')]
    all_primary=f[(f.variant=='primary')&(f.status=='fit_success')]
    assert b.successful.le(spec['bootstrap_draws']).all()
    assert b.loc[b.interval_admitted.eq(1),'successful'].ge(spec['bootstrap_success_minimum']).all()
    assert f.measurement_mode.eq('nominal_development').all() and f.real_input_verified.eq(0).all()
    assert f.specification_id.eq(spec['specification_id']).all(), 'Aggregate specification does not match current manuscript specification'
    assert set(['normalizer_scope','coefficient_year_min','coefficient_year_max']).issubset(data['deep_prediction_profiles'].columns), 'Old diagnostic schema cannot supply this manuscript'
    assert data['deep_prediction_profiles'].normalizer_scope.dropna().eq('pooled_dynamic').all()
    keys={
      'nominal_sample_support':['sector','method','sample_id'],
      'nominal_coefficients':['sector','method','sample_id','variant'],
      'nominal_bootstrap_summary':['sector','method','sample_id'],
      'nominal_distribution_quantiles':['sector','method','sample_id','measure'],
      'nominal_pairwise_comparisons':['sector','sample_id','method_a','method_b','comparison_scope']}

    for name in data:
        if name.startswith('deep_'):
            keys[name]=[k for k in ['sector','method','sample_id','source_year','year_from','year_to','input','scale','fit_scope','measure','stage','variant','example_type','bin_lower','bin_upper'] if k in data[name].columns]

    def operand(name,row,field):
        index=int(row.name)
        return {'source_file':inputs[name]['path'],'source_sha256':inputs[name]['sha256'],'csv_line':index+2,'row_key':{k:clean(row[k]) for k in keys[name]},'field':field,'value':clean(row[field]),'source_text':raw[name][index][field]}
    def emit(bucket,anchor,value,operands,fmt='.3f',scale=1,operation='identity',display=None):
        if display is None:
            display='---' if pd.isna(value) else format(float(value)*scale,fmt)
        traces[bucket].append({'anchor':anchor,'operation':operation,'operands':operands,'unrounded_value':clean(value),'scale':scale,'format':fmt,'printed_string':display})
        return display
    def cell(name,row,field,anchor,fmt='.3f',display=None):
        if field in {'source_year','year','year_from','year_to','coefficient_year','normalizer_year'}: fmt='.0f'
        return emit('table_cells',anchor,row[field],[operand(name,row,field)],fmt=('text-map' if display is not None else fmt),operation=('label mapping' if display is not None else 'identity'),display=display)
    def val(name,row,field,anchor,fmt='.3f'):
        return emit('numerical_prose',anchor,row[field],[operand(name,row,field)],fmt=fmt)
    def deriv(bucket,anchor,rows,fields,operation,value,fmt='.3f'):
        return emit(bucket,anchor,value,[operand(name,row,field) for name,row in rows for field in fields],fmt=fmt,operation=operation)
    def specval(field,anchor,fmt=',.0f'):
        value=spec[field]
        return emit('numerical_prose',anchor,value,[{'source_file':str(specpath),'source_sha256':sha(specpath),'json_pointer':'/'+field,'value':value}],fmt=fmt)
    def declared(anchor,field,printed):
        return emit('table_cells',anchor,spec[field],[{'source_file':str(specpath),'source_sha256':sha(specpath),'json_pointer':'/'+field,'value':spec[field]}],fmt='declared-definition paraphrase',display=printed)
    def one(name,**query):
        g=data[name]
        for k,v in query.items(): g=g[g[k].eq(v)]
        assert len(g)==1,(name,query)
        return g.iloc[0]
    def table(file,caption,label,columns,head,rows,notes,source=SOURCE):
        if len(rows)>24:
            header=' & '.join(head)+' \\\\\n'
            text='\\begingroup\\fontsize{9.2}{11.2}\\selectfont\n\\setlength{\\tabcolsep}{4.3pt}\\renewcommand{\\arraystretch}{1.1}\n'
            text+='\\begin{longtable}{'+columns+'}\n\\caption{'+caption+'}\\GOneLabel{tab:'+label+'}\\\\\n\\toprule\n'+header+'\\midrule\n\\endfirsthead\n'
            text+='\\multicolumn{'+str(len(head))+'}{l}{\\textit{'+caption+' (continued)}}\\\\\n\\toprule\n'+header+'\\midrule\n\\endhead\n'
            text+='\\midrule\n\\multicolumn{'+str(len(head))+'}{r}{\\textit{Continued on next page}}\\\\\n\\endfoot\n\\bottomrule\n\\endlastfoot\n'
            text+='\n'.join(' & '.join(row)+' \\\\' for row in rows)
            text+='\n\\end{longtable}\n\\endgroup\n\\appnotes{'+notes+'}{'+source+'}\n'
            (P/'tables'/file).write_text(text)
            return
        text='\\begin{table}[H]\n\\centering\n\\caption{'+caption+'}\n\\GOneLabel{tab:'+label+'}\n\\begingroup\\fontsize{9.2}{11.2}\\selectfont\n\\setlength{\\tabcolsep}{4.3pt}\\renewcommand{\\arraystretch}{1.1}\n\\begin{tabular}{'+columns+'}\n\\toprule\n'+' & '.join(head)+' \\\\\n\\midrule\n'
        text+='\n'.join(' & '.join(row)+' \\\\' for row in rows)
        text+='\n\\bottomrule\n\\end{tabular}\n\\endgroup\n\\appnotes{'+notes+'}{'+source+'}\n\\end{table}\n'
        (P/'tables'/file).write_text(text)

    # Definitions bind to the maintained specification; no historic documentary gate is silently reinstated.
    definitions=[('Outcome $y$','outcome','Log positive reported nominal value added'),('Labor $l$','labor','Log positive reported workers'),('Capital $k$','capital','Log positive preceding-calendar-year stock, assuming year-end timing'),('OP proxy $i$','op_proxy','Log positive reported additions, assuming investment meaning'),('Other proxies $m$','lp_acf_wrdg_proxy','Log positive reported raw/supporting materials')]
    rows=[]
    for i,(label,field,printed) in enumerate(definitions):
        for j,text in enumerate([label,printed]): declared(f'measurement:r{i+1}:c{j+1}',field,text)
        rows.append([label,printed])
    table('measurement.tex','Variables in the production equation','measurement','lp{10.0cm}',['Measure','Construction'],rows,'We retain each variable\'s reported unit scale. Capital at the beginning of a production year uses the preceding calendar-year total under the year-end-stock assumption. Reported additions separately supply assumed investment. We require positive values and exact timing, preserve the narrower documentary flags, and apply no price adjustment.',"G0.5 v3 development fields and the assumed-capital nominal specification; authors' construction.")
    rows=[]
    observed_sectors=sorted(supp.sector.unique())
    for sector in observed_sectors:
        lp=one('nominal_sample_support',sector=sector,method='lp',sample_id='own');op=one('nominal_sample_support',sector=sector,method='op',sample_id='own');common=one('nominal_sample_support',sector=sector,method='lp',sample_id='common')
        rows.append([cell('nominal_sample_support',lp,'sector',f'support:{sector}:sector',display=sector),*[cell('nominal_sample_support',r,col,f'support:{sector}:{tag}',',.0f') for r,col,tag in [(lp,'N_domain','materials_domain'),(lp,'N_dynamic','materials_dynamic'),(common,'N_dynamic','common_dynamic'),(op,'N_domain','investment_domain'),(op,'N_selection','selection'),(op,'N_disappearance','disappearance')]]])
    table('support.tex','Input and consecutive-year samples by sector','support','lrrrrrr',['Code',r'\makecell{Materials\\inputs}',r'\makecell{Materials\\pairs}',r'\makecell{Four-method\\pairs}',r'\makecell{Investment\\inputs}',r'\makecell{OP risk\\intervals}',r'\makecell{Fictional\\disappearances}'],rows,'The materials domain supplies LP, ACF and Wooldridge before method-specific numerical checks. All counts pool supported calendar years. Four-method eligibility additionally requires the investment proxy before taking exact lags. The OP risk sample includes eligible prior-year state and proxy information, including assignments that disappear, separately from the continuing production sample. Counts are observations or intervals, not unique companies. We retain every observed sector, including those without a successful fit.')
    rows=[]
    for _,rr in all_primary.sort_values(['sample_id','sector','method']).iterrows():
        sector,m,sample=rr.sector,rr.method,rr.sample_id
        br=one('nominal_bootstrap_summary',sector=sector,method=m,sample_id=sample)
        anchor=f'coeff:{sample}:{sector}:{m}'
        vals=[cell('nominal_coefficients',rr,'sample_id',anchor+':sample',display=sample.capitalize()),cell('nominal_coefficients',rr,'sector',anchor+':sector',display=sector),cell('nominal_coefficients',rr,'method',anchor+':method',display=METHOD[m]),cell('nominal_coefficients',rr,'N',anchor+':N',',.0f')]
        for term,lo,hi in [('L','ci_L_lo','ci_L_hi'),('K','ci_K_lo','ci_K_hi')]:
            point=cell('nominal_coefficients',rr,'beta_'+term,anchor+':beta'+term,'.4f');low=cell('nominal_bootstrap_summary',br,lo,anchor+':'+lo,'.4f');high=cell('nominal_bootstrap_summary',br,hi,anchor+':'+hi,'.4f')
            vals.append(r'\makecell{'+point+r'\\{['+low+', '+high+']}}')
        vals.append(cell('nominal_coefficients',rr,'returns_to_scale',anchor+':RTS','.4f'))
        rows.append(vals)
    table('coefficients.tex','Production coefficients and bootstrap intervals','coefficients','lllrrrr',['Sample','Code','Method','$N$',r'$\widehat\beta_L$',r'$\widehat\beta_K$',r'$\widehat\beta_L+\widehat\beta_K$'],rows,'Each row uses one sector and sample with constant coefficients over its eligible years, the primary cubic approximation and the ordinary-least-squares starting rule. Own samples use each method\'s required inputs; common samples require all four methods\' inputs before taking exact lags. Brackets contain admitted 95 percent percentile intervals from whole-history bootstrap resamples. A dash means an interval is unavailable under the minimum-success rule. The method tables show standard errors and successful draw counts. We retain the original estimates, without imposing coefficient signs or returns to scale after seeing the results.')
    rows=[]
    for sample in ['own','common']:
        for m in METHOD:
            rr=one('nominal_distribution_quantiles',sector='ALL',method=m,sample_id=sample,measure='centered')
            rows.append([cell('nominal_distribution_quantiles',rr,'sample_id',f'distribution:{sample}:{m}:sample',display=sample.capitalize()),cell('nominal_distribution_quantiles',rr,'method',f'distribution:{sample}:{m}:method',display=METHOD[m]),*[cell('nominal_distribution_quantiles',rr,c,f'distribution:{sample}:{m}:{c}',',.0f' if c=='N' else '.3f') for c in ['N','sd','p1','p25','p50','p75','p99']]])
    table('distribution.tex','Pooled centered residual distributions','distribution','llrrrrrrr',['Sample','Method','$N$','SD','P1','P25','Median','P75','P99'],rows,'Each distribution uses the named method\'s pooled dynamic estimation records. We subtract its own sector-method-sample fitted mean. The pooled fit mean is therefore zero when the group is nonempty. An own-sample distribution can use different records from another method; the common rows use the separate common-input estimation specification. Quantiles and standard deviations describe dispersion across observations, not coefficient uncertainty. Zero counts and unavailable numeric summaries remain distinct.')
    rows=[]
    for sample in ['own','common']:
        for ma,mb in [('op','lp'),('op','acf'),('op','wrdg'),('lp','acf'),('lp','wrdg'),('acf','wrdg')]:
            rr=one('nominal_pairwise_comparisons',sector='ALL',sample_id=sample,method_a=ma,method_b=mb,comparison_scope='pairwise')
            anchor=f'pairs:{sample}:{ma}:{mb}'
            rows.append([cell('nominal_pairwise_comparisons',rr,'sample_id',anchor+':sample',display=sample.capitalize()),cell('nominal_pairwise_comparisons',rr,'method_a',anchor+':a',display=METHOD[ma])+r'--'+cell('nominal_pairwise_comparisons',rr,'method_b',anchor+':b',display=METHOD[mb]),*[cell('nominal_pairwise_comparisons',rr,c,anchor+':'+c,',.0f' if c.startswith('N_') else '.3f') for c in ['N_compared','N_four_method','pearson_centered','rho_spearman']]])
    table('rank_comparison.tex','Residual agreement on shared estimation records','rank-comparison','llrrrr',['Sample','Methods',r'\makecell{Pairwise\\records}',r'\makecell{Four-method\\records}',r'\makecell{Pearson\\correlation}',r'\makecell{Spearman\\correlation}'],rows,'We match primary dynamic-fit predictions to the same original company-year record. Each row compares the pair\'s shared observations within the named estimation specification. Own and common rows can use different coefficient vectors as well as different records. Pearson and Spearman correlations use centered logs, with average ranks for ties. The four-method count describes the intersection of all four fitted prediction sets, which need not equal a pairwise intersection. A dash means the correlation is unavailable.')
    sens=[];rows=[]
    for m in methods:
        cols=[]
        for variants,term,fmt in [(['low_l','high_l'],'L','.2e'),(['low_l','high_l'],'K','.2e'),(['poly2'],'L','.4f'),(['poly2'],'K','.4f')]:
            operands=[];differences=[]
            for _,pr in primary.loc[primary.method.eq(m)].iterrows():
                for variant in variants:
                    sr=one('nominal_coefficients',sector=pr.sector,method=m,sample_id='own',variant=variant)
                    if sr.status!='fit_success': continue
                    operands += [operand('nominal_coefficients',pr,'beta_'+term),operand('nominal_coefficients',sr,'beta_'+term)]
                    differences.append(abs(sr['beta_'+term]-pr['beta_'+term]))
            if not differences:
                cols.append('---');continue
            maximum=max(differences)
            cols.append(emit('table_cells',f'sensitivity:{m}:{variants[0]}:{term}',maximum,operands,fmt=fmt,operation='maximum absolute paired sensitivity minus primary, across displayed sectors'))
            sens.append((m,variants[0],term,maximum,operands))
        rr=primary.loc[primary.method.eq(m)].iloc[0]
        rows.append([cell('nominal_coefficients',rr,'method',f'sensitivity:{m}:method',display=METHOD[m]),*cols])
    table('sensitivity.tex','Coefficient changes under alternative estimation choices','sensitivity','lrrrr',['Method',r'\makecell{Starts:\\labor}',r'\makecell{Starts:\\capital}',r'\makecell{Quadratic:\\labor}',r'\makecell{Quadratic:\\capital}'],rows,'Each entry is the maximum absolute difference from the primary coefficient over sectors with successful own-sample fits in both compared variants. Alternative deterministic starts are $(0.3,0.7)$ and $(0.7,0.3)$ with the cubic proxy. The quadratic comparison retains the ordinary-least-squares start and changes proxy degree to two. OP, LP and ACF retain the separate cubic transition; Wooldridge changes both current and lagged proxy bases. The input sample and median translations remain fixed. Missing alternatives are not treated as zero changes; the attempt tables retain their outcomes.')

    n={}
    source_manifest=ROOT/'data/g0-5-development/manifest.json';source_meta=json.loads(source_manifest.read_text())
    n['source_records']=emit('numerical_prose','source:records',source_meta['rows'],[{'source_file':str(source_manifest),'source_sha256':sha(source_manifest),'json_pointer':'/rows','value':source_meta['rows']}],fmt=',.0f')
    n['sectors']=deriv('numerical_prose','sample:sector_count',[('nominal_coefficients',r) for _,r in primary.iterrows()],['sector'],'count distinct admitted sector codes',len(sectors),',.0f')
    n['primary']=deriv('numerical_prose','sample:primary_count',[('nominal_coefficients',r) for _,r in primary.iterrows()],['status'],'count admitted own-domain primary coefficient rows',len(primary),',.0f')
    n['common_primary']=deriv('numerical_prose','sample:common_primary_count',[('nominal_coefficients',r) for _,r in common_primary.iterrows()],['status'],'count successful fixed-variant coefficient rows',len(common_primary),',.0f')
    n['allfits']=deriv('numerical_prose','sensitivity:fit_count',[('nominal_coefficients',r) for _,r in f[f.status.eq('fit_success')].iterrows()],['status'],'count successful fixed-variant coefficient rows',int(f.status.eq('fit_success').sum()),',.0f')
    ownboot=b[b.sample_id.eq('own')]
    commonboot=b[b.sample_id.eq('common')]
    n['draws']=specval('bootstrap_draws','inference:draws')
    n['drawtotal']=deriv('numerical_prose','inference:total_resamples',[('nominal_bootstrap_summary',r) for _,r in ownboot.iterrows()],['successful'],'sum successful resamples over all admitted primary cells',ownboot.successful.sum(),',.0f')
    n['common_drawtotal']=deriv('numerical_prose','inference:common_total_resamples',[('nominal_bootstrap_summary',r) for _,r in commonboot.iterrows()],['successful'],'sum successful resamples over all admitted primary cells',commonboot.successful.sum(),',.0f')
    for field,key in [('minimum_first_stage_N','minfirst'),('minimum_dynamic_N','mindyn'),('minimum_dynamic_firms','minfirm'),('minimum_OP_selection_N','minrisk'),('minimum_OP_selection_exits','minexit'),('minimum_OP_selection_continuers','mincontinue'),('bootstrap_success_minimum','minsuccess')]: n[key]=specval(field,'threshold:'+field)
    for field in ['history_years','production_years','dynamic_years']:
        n[field]=emit('numerical_prose','timing:'+field,spec[field],[{'source_file':str(specpath),'source_sha256':sha(specpath),'json_pointer':'/'+field,'value':spec[field]}],fmt='declared design',display=year_text(spec[field]))

    def method_count(m):
        part=primary[primary.method.eq(m)]
        return deriv('numerical_prose','sample:'+m+':sectors',[('nominal_coefficients',r) for _,r in part.iterrows()],['sector'],'count distinct admitted sector codes',part.sector.nunique(),',.0f')
    method_counts={m:method_count(m) for m in METHOD}
    coverage=[]
    for m in METHOD:
        rr=one('nominal_distribution_quantiles',sector='ALL',method=m,sample_id='own',measure='centered')
        count=val('nominal_distribution_quantiles',rr,'N','sample:'+m+':fitN',',.0f')
        estimate_noun='estimate' if int(primary.method.eq(m).sum())==1 else 'estimates'
        coverage.append(f'{METHOD[m]} has {method_counts[m]} accepted sector {estimate_noun} and {count} dynamic company-year observations')
    n['support_results']='; '.join(coverage)+'. These counts describe each method\'s own sample. Repeated annual observations from a fictional company remain distinct records. We keep failures of the sample and numerical requirements visible rather than borrowing coefficients from another sector.'
    n['intro_results']=f"Using the observations required by each method, we obtain {n['primary']} main sector estimates: LP in {method_counts['lp']} sectors, ACF in {method_counts['acf']}, Wooldridge in {method_counts['wrdg']} and OP in {method_counts['op']}. These estimates span {n['sectors']} manufacturing groups. Requiring inputs suitable for all four methods gives {n['common_primary']} separate common-sample estimates. We compare the methods on shared records and distinguish that comparison from the coverage of their broader samples."
    n['opening_answer']='We compare the value manufacturers report creating after accounting for the workers and capital they use. The methods support different comparisons: some cover more industry groups, while a comparison shared by all four uses a much smaller set of observations. We therefore examine the estimated input contributions alongside the records on which the methods can be compared.'

    coefficient_parts=[]
    for field,title in [('beta_L','labor'),('beta_K','capital')]:
        low=primary.loc[primary[field].idxmin()];high=primary.loc[primary[field].idxmax()]
        lo=val('nominal_coefficients',low,field,'coeff:range:'+field+':min','.4f');hi=val('nominal_coefficients',high,field,'coeff:range:'+field+':max','.4f')
        coefficient_parts.append(f'the {title} coefficient ranges from {lo} ({METHOD[low.method]}, sector {low.sector}) to {hi} ({METHOD[high.method]}, sector {high.sector})')
    n['coefficient_results']='Across own-sample estimates, '+'; '.join(coefficient_parts)+'. These ranges describe variation across both methods and sectors. They do not isolate a method effect unless we also fix the sector and comparison sample. We retain the estimated signs rather than constraining the coefficients after seeing them.'
    uncertainty_synthesis='The bootstrap summaries describe additional variation in the estimated input contributions when the sample histories change.'
    if ownboot.se_L.notna().any():
        uncertain=ownboot.loc[ownboot.se_L.idxmax()]
        se=val('nominal_bootstrap_summary',uncertain,'se_L','coefficient:largest_labor_se','.4f')
        lo=val('nominal_bootstrap_summary',uncertain,'ci_L_lo','coefficient:largest_labor_se:lo','.4f')
        hi=val('nominal_bootstrap_summary',uncertain,'ci_L_hi','coefficient:largest_labor_se:hi','.4f')
        n['coefficient_results']+=f" The largest labor standard error is {se}, for {METHOD[uncertain.method]} in sector {uncertain.sector}; its admitted percentile interval is [{lo}, {hi}]. Standard errors measure dispersion across the resampled estimates, while percentile intervals can leave a small number of extreme resamples outside their endpoints. We examine both summaries and the full coefficient tails in the method chapters."
        uncertainty_synthesis=f"Uncertainty also differs across equations: the largest labor bootstrap standard error is {se}, for {METHOD[uncertain.method]} in sector {uncertain.sector}."

    distribution_parts=[]
    for m in methods:
        rr=one('nominal_distribution_quantiles',sector='ALL',method=m,sample_id='own',measure='centered')
        n['sd_'+m]=val('nominal_distribution_quantiles',rr,'sd','distribution:sd_'+m,'.3f')
        distribution_parts.append(METHOD[m]+' '+n['sd_'+m])
    n['distribution_results']='The standard deviations of own-sample centered residuals are '+', '.join(distribution_parts)+' log points. A narrower distribution leaves less variation after the fitted input contributions, but does not establish better measurement of physical productivity. These own-sample summaries also differ in sector and observation coverage. The common-sample rows and matched comparisons help separate that coverage issue from differences between fitted equations.'
    valid=pairs.query("sector == 'ALL' and sample_id == 'own' and comparison_scope == 'pairwise'").dropna(subset=['rho_spearman'])
    pair_parts=[]
    for _,rr in valid.iterrows():
        pair=f'{METHOD[rr.method_a]}--{METHOD[rr.method_b]}'
        rho=val('nominal_pairwise_comparisons',rr,'rho_spearman',f'pairs:own:{rr.method_a}:{rr.method_b}:rho','.3f')
        nn=val('nominal_pairwise_comparisons',rr,'N_compared',f'pairs:own:{rr.method_a}:{rr.method_b}:N',',.0f')
        pair_parts.append(f'{pair}: {rho} on {nn} shared records')
    n['pair_results']='The own-sample rank correlations are '+('; '.join(pair_parts) if pair_parts else 'unavailable under the required shared-record comparison')+'. Each pair holds its records fixed, but different pairs can use different intersections. The common-sample rows compare the separately fitted common-input specifications; we do not equate those coefficients with the own-sample coefficients. Correlation measures agreement, not which estimator best measures physical productivity.'
