"""Write acceptance, quality review and README from matching completed evidence."""
from pathlib import Path

import hashlib
import json

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[1]
METHODS = {'op': 'Olley–Pakes', 'lp': 'Levinsohn–Petrin', 'acf': 'Ackerberg–Caves–Frazer', 'wrdg': 'Wooldridge'}
SAMPLES = ('own', 'common')


def read(path):
    return json.loads(path.read_text())


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def years(values):
    return ', '.join(str(int(v)) for v in sorted(set(values))) or 'None'


def table(headers, rows):
    return '| ' + ' | '.join(headers) + ' |\n| ' + ' | '.join(['---'] * len(headers)) + ' |\n' + '\n'.join('| ' + ' | '.join(str(v) for v in row) + ' |' for row in rows)


def summarize(fits, boot, pairs, profiles):
    primary = fits[fits.variant.eq('primary')]
    good = primary[primary.status.eq('fit_success')]
    coverage = profiles[profiles.sector.eq('ALL') & profiles.fit_scope.eq('all') & profiles.measure.eq('raw') & profiles.source_year.gt(0)]
    model_rows, stability = [], []
    for method in METHODS:
        for sample in SAMPLES:
            all_cells = primary[primary.method.eq(method) & primary.sample_id.eq(sample)]
            cells = good[good.method.eq(method) & good.sample_id.eq(sample)]
            b = boot[boot.method.eq(method) & boot.sample_id.eq(sample)]
            annual = coverage[coverage.method.eq(method) & coverage.sample_id.eq(sample)]
            assert int(annual.N_predicted.sum()) == int(cells.N_domain.sum())
            assert int(annual.N_in_fit.sum()) == int(cells.N_dynamic.sum())
            model_rows.append({'method': method, 'sample': sample, 'primary_models': len(cells),
                'successful_sectors': sorted(cells.sector.tolist()),
                'skipped_primary_models': int(all_cells.status.eq('skipped_support').sum()),
                'failed_primary_models': int(all_cells.status.eq('fit_failed').sum()),
                'dynamic_rows': int(cells.N_dynamic.sum()), 'prediction_rows': int(cells.N_domain.sum()),
                'dynamic_years': sorted(int(y) for y in annual.loc[annual.N_in_fit.gt(0), 'source_year']),
                'prediction_years': sorted(int(y) for y in annual.loc[annual.N_predicted.gt(0), 'source_year']),
                'bootstrap_interval_models': int(b.interval_admitted.eq(1).sum()),
                'bootstrap_withheld_models': int(b.interval_admitted.ne(1).sum())})
            variants = fits[fits.method.eq(method) & fits.sample_id.eq(sample) & fits.variant.isin(['low_l', 'high_l']) & fits.status.eq('fit_success')]
            matched = variants.merge(cells[['sector', 'beta_L', 'beta_K']], on='sector', suffixes=('', '_primary'), validate='many_to_one')
            if len(matched):
                stability.append([METHODS[method], sample, len(matched), f'{(matched.beta_L - matched.beta_L_primary).abs().max():.4f}', f'{(matched.beta_K - matched.beta_K_primary).abs().max():.4f}'])
    intersections = {}
    for sample in SAMPLES:
        p = pairs[pairs.sector.eq('ALL') & pairs.sample_id.eq(sample) & pairs.comparison_scope.eq('four_method')]
        assert len(p) == 6 and p.N_compared.nunique() == 1
        intersections[sample] = int(p.N_compared.iloc[0])
    return {'primary_models': len(good), 'successful_all_variant_models': int(fits.status.eq('fit_success').sum()),
        'successful_sensitivity_models': int((fits.variant.ne('primary') & fits.status.eq('fit_success')).sum()),
        'obligation_cells': len(fits), 'attempted_cells': int(fits.attempted.eq(1).sum()),
        'failed_cells': int(fits.status.eq('fit_failed').sum()), 'skipped_cells': int(fits.status.eq('skipped_support').sum()),
        'bootstrap_requested_total': int(boot.requested.sum()), 'bootstrap_successful_total': int(boot.successful.sum()),
        'bootstrap_failed_calls': int(boot.rc.ne(0).sum()),
        'bootstrap_failed_retained_draws': int((boot.loc[boot.rc.eq(0), 'requested'] - boot.loc[boot.rc.eq(0), 'successful']).sum()),
        'bootstrap_interval_models': int(boot.interval_admitted.eq(1).sum()),
        'bootstrap_withheld_models': int(boot.interval_admitted.ne(1).sum()),
        'method_sample_results': model_rows, 'four_method_shared_dynamic_rows': intersections,
        'annual_coverage': coverage[['method', 'sample_id', 'source_year', 'N_source', 'N_domain', 'N_predicted', 'N_in_fit', 'N_outside_fit']].to_dict('records'),
        'stability_rows': stability}

## Reproduce

Use a new external run ID and one Stata worker. From the repository root, with the declared Python dependencies, Stata SE 17, XeLaTeX and actual Times New Roman installed:

```sh
OPENBLAS_NUM_THREADS=1 OMP_NUM_THREADS=1 python3 work-items/G1/run.py --run-id a-new-run-id --publish
```

Before the diagnostic run, preserve the previous diagnostic contract with its external run and register `base_run` in `deep-diagnostics-contract.json` as the exact new base directory name. Keep its scientific definitions unchanged. The independent validator rejects a run whose name differs from this explicit source binding. Freeze the corrected contract and code before execution.

```sh
stata-se -b do work-items/G1/stata/G1.do /absolute/repository /absolute/new-diagnostics-run diagnostics /absolute/base-run
python3 work-items/G1/validate_deep.py --base /absolute/base-run --run /absolute/new-diagnostics-run
```

After diagnostic validation, retain the former release with its external run. Copy only the CSVs enumerated in `deep-diagnostics-contract.json`, matching manifests, validation receipts and quiet execution log into `deep-release`. Keep record-level intermediates external. Then rebuild the publication:

```sh
python3 work-items/G1/build_contracts.py
python3 work-items/G1/build_report.py
python3 work-items/G1/publication/plot_deep_figures.py --base work-items/G1/release --deep work-items/G1/deep-release
python3 work-items/G1/publication/build_appendix.py --results work-items/G1/release
sh work-items/G1/publication/build.sh
python3 work-items/G1/validate_publication_bindings.py
python3 harness/manage.py validate
```

Inspect the compiled pages and regenerate exact-file visual and source-review receipts before running `python3 work-items/G1/build_closeout.py`. The builder checks matching base-run, diagnostic, specification, current Luna/max review and PDF identities before generating this README, acceptance and quality review. Compilation alone does not establish numerical or visual acceptance.

Record-level files, including `analysis/G1_main_with_tfp.dta`, remain in `<checkout>-g1-runs/<run-id>/`. The derivative preserves all original adapter fields and attaches distinct `dev_nominal_*` predictions and flags. The shared adapter checks the pinned G0.5 package. [Measurement modes](stata/methods/MEASUREMENT-MODES.md) preserve the distinction from actual finalisation.
