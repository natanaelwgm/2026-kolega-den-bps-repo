q_it = alpha_i + lambda_t + sum_{t != 2019} beta_t D_i,pre 1[year=t] + u_it; q^L_it = L_it/L_i,2019 - 1; q^Y_it = Y_it/Y_i,2019 - 1; q^S_it = verified_alive_it
